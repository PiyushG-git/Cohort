- depoloymet.yml(manage the manage the pod)  
    Deployment simple

apiVersion: apps/v1   // name of version
kind: Deployment
metadata:
  name: express-deployment             // deployment name
spec:
  replicas: 3          // number of pods
  selector:
    matchLabels:
      app: main-server
  template:
    metadata:
      labels:
        app: main-server               //pod name
    spec:
      containers:                          // each container having this
      - name: main-server                      // container name
        image: express_cohort_2:latest         //image name
        imagePullPolicy: Always
        ports:
        - containerPort: 3000         //trafic on this port
        resources:
          limits:
            memory: "128Mi" # RAM 128 megabytes
            cpu: "500m" # CPU 500 millicores (0.5 CPU core)
            # 1 core = 1000m (1000 millicores)
          requests:
            memory: "64Mi" # RAM 64 megabytes
            cpu: "250m" # CPU 250 millicores (0.25 CPU core)

service.yml  - (service complete)

kind: Service
apiVersion: v1
metadata:
  name:  main-server-service    // service name
spec:
  selector:
    app:  main-server    // same pod name
  type: ClusterIP             //cluster iP , loadbalancer
  ports:
  - name:  name-of-the-port
    port:  80
    targetPort:  3000     // port target

ingress.yml

kubectl apply -f \
  https://raw.githubusercontent.com/kubernetes/ingress-nginx/controller-v1.12.1/deploy/static/provider/cloud/deploy.yaml
   
   to install the nginx
   then write rules

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: express-ingress            //name of ingress
  labels:
    app.kubernetes.io/name: express-ingress
spec:
  ingressClassName: nginx                  //imp to mention
  rules:
    - http:
        paths:
        - pathType: Prefix
          path: "/"
          backend:
            service:
              name: express-service    // same as service
              port: 
                number: 80

#-before this we have to enable the metric server
# Option A: Official manifest with kubelet insecure TLS patch (recommended for Docker Desktop)
kubectl apply -f https://github.com/kubernetes-sigs/metrics-server/releases/latest/download/components.yaml

# Docker Desktop uses a self-signed cert — patch metrics-server to skip TLS verify:
kubectl patch deployment metrics-server -n kube-system \
  --type=json \
  -p='[{"op":"add","path":"/spec/template/spec/containers/0/args/-","value":"--kubelet-insecure-tls"}]'

# Wait for it to be ready (~30 seconds)
kubectl rollout status deployment/metrics-server -n kube-system

# Verify it works — you should see CPU and Memory columns
kubectl top nodes
kubectl top pods
